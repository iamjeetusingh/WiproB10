mkdir -p /usr/local/nagiosxi/var/certs

openssl req -x509 -nodes -days 365 -newkey rsa:2048 \
-keyout /usr/local/nagiosxi/var/certs/nagiosxi.key \
-out /usr/local/nagiosxi/var/certs/nagiosxi.crt

chmod 600 /usr/local/nagiosxi/var/certs/nagiosxi.key
chmod 644 /usr/local/nagiosxi/var/certs/nagiosxi.crt

systemctl restart httpd
